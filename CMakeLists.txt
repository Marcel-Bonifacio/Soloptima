# Alternative build:  cmake -B build -DCMAKE_BUILD_TYPE=Release && cmake --build build   (`make` is the main build).
# Needs libcurl on macOS / Linux; Windows uses the built-in WinHTTP. Builds the command line only (the app is src-tauri).
cmake_minimum_required(VERSION 3.16)
project(Soloptima CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(Threads REQUIRED)
file(GLOB SRC src/*.cpp)
add_executable(soloptima ${SRC})
target_link_libraries(soloptima PRIVATE Threads::Threads)
if(WIN32)
  target_compile_definitions(soloptima PRIVATE NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_libraries(soloptima PRIVATE ws2_32 winhttp shell32 ole32 uuid advapi32 wintrust crypt32)
else()
  find_package(CURL REQUIRED)
  target_link_libraries(soloptima PRIVATE CURL::libcurl)
  if(APPLE)
    target_link_libraries(soloptima PRIVATE "-framework Security" "-framework CoreFoundation")
  endif()
endif()
