# Alternative build:  cmake -B build -DCMAKE_BUILD_TYPE=Release && cmake --build build   (`make` is the main build).
# Needs libcurl on macOS / Linux. On Windows it uses the built-in WinHTTP and the WebView2 SDK in build/webview2, which
# packaging/windows/fetch-webview2.sh fetches; copy build/webview2/WebView2Loader.dll next to the program.
cmake_minimum_required(VERSION 3.16)
project(Soloptima CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(Threads REQUIRED)
file(GLOB SRC src/*.cpp)
if(APPLE)
  enable_language(OBJCXX)
  list(APPEND SRC src/app_mac.mm)
  set_source_files_properties(src/app_mac.mm PROPERTIES COMPILE_FLAGS -fobjc-arc)
endif()
add_executable(soloptima ${SRC})
target_link_libraries(soloptima PRIVATE Threads::Threads)
if(WIN32)
  set_target_properties(soloptima PROPERTIES WIN32_EXECUTABLE ON)
  target_compile_definitions(soloptima PRIVATE NOMINMAX _WIN32_WINNT=0x0A00)
  target_include_directories(soloptima SYSTEM PRIVATE ${CMAKE_SOURCE_DIR}/build/webview2/include)
  target_link_libraries(soloptima PRIVATE ws2_32 winhttp shell32 ole32 uuid advapi32 wintrust crypt32)
else()
  find_package(CURL REQUIRED)
  target_link_libraries(soloptima PRIVATE CURL::libcurl)
  if(APPLE)
    target_link_libraries(soloptima PRIVATE "-framework Security" "-framework CoreFoundation" "-framework Cocoa" "-framework WebKit")
  endif()
endif()
